if (UnityEditor.EditorApplication.isPlayingOrWillChangePlaymode || UnityEditor.EditorApplication.isCompiling || UnityEditor.EditorApplication.isUpdating)
    return new { skipped = true, reason = "Editor is busy; no state was changed." };
var sceneBefore = UnityEngine.SceneManagement.SceneManager.GetActiveScene();
var dirtyBefore = sceneBefore.isDirty;
var definition = UnityEditor.AssetDatabase.LoadAssetAtPath<PlayerClassDefinition>("Assets/Resources/Classes/swordsman.asset");
var outputs = new System.Collections.Generic.List<object>();
var folder = "E:/project/RPG-Silent/Docs/UIDesign/2026-10-29-functional-v5/assets/models";
System.IO.Directory.CreateDirectory(folder);
foreach (var state in new [] { "current", "trial", "weapon", "empty" })
{
    var equipped = new System.Collections.Generic.List<ItemDefinition>();
    if (state != "empty")
        foreach (var id in new [] { "crystal-head", state == "trial" ? "warden-chest" : "crystal-chest", "crystal-feet", "crystal-ring" })
            equipped.Add(UnityEditor.AssetDatabase.LoadAssetAtPath<ItemDefinition>("Assets/Resources/Items/" + id + ".asset"));
    var gender = definition.GetVisualGender(PlayerGender.Male, equipped.Count > 0);
    var preview = new UnityEditor.PreviewRenderUtility(true);
    UnityEngine.Texture2D pixels = null;
    ModularArmorAppearance armor = null;
    try
    {
        // The model is instantiated directly into a preview scene, never the user's active scene.
        var model = preview.InstantiatePrefabInScene(definition.GetModel(gender));
        model.transform.position = UnityEngine.Vector3.zero;
        var animator = model.GetComponent<UnityEngine.Animator>();
        var controller = animator.runtimeAnimatorController;
        UnityEngine.AnimationClip idle = null;
        if (controller != null)
            idle = System.Linq.Enumerable.FirstOrDefault(controller.animationClips, c => c.name.ToLowerInvariant().Contains("unarmed") && c.name.ToLowerInvariant().Contains("idle"));
        animator.runtimeAnimatorController = null;
        animator.applyRootMotion = false; animator.fireEvents = false; animator.Rebind();
        var profile = model.GetComponent<CharacterRigProfile>();
        armor = new ModularArmorAppearance(model.transform, profile, gender);
        armor.Apply(equipped);
        (idle != null ? idle : definition.PreviewIdle).SampleAnimation(model, .3f);
        var weaponItem = UnityEditor.AssetDatabase.LoadAssetAtPath<ItemDefinition>("Assets/Resources/Items/" + (state == "weapon" ? "rare-sword" : "training-sword") + ".asset");
        var weapon = weaponItem.Weapon;
        var bone = animator.GetBoneTransform(weapon.SheathBone);
        if (bone != null && weapon.ModelPrefab != null)
        {
            var prop = UnityEngine.Object.Instantiate(weapon.ModelPrefab, bone);
            prop.transform.localPosition = profile.MountPosition(weapon, false);
            prop.transform.localRotation = profile.MountRotation(weapon, false);
            prop.transform.localScale = profile.Scale(weapon.ModelScale);
        }
        model.transform.rotation = UnityEngine.Quaternion.Euler(0, 165, 0);
        var renderers = System.Linq.Enumerable.ToArray(System.Linq.Enumerable.Where(model.GetComponentsInChildren<UnityEngine.Renderer>(), r => r.enabled && r.gameObject.activeInHierarchy));
        var bounds = renderers[0].bounds;
        foreach (var renderer in renderers) bounds.Encapsulate(renderer.bounds);
        var camera = preview.camera;
        camera.clearFlags = UnityEngine.CameraClearFlags.SolidColor;
        camera.backgroundColor = new UnityEngine.Color(0, 0, 0, 0);
        camera.fieldOfView = 30; camera.nearClipPlane = .05f; camera.farClipPlane = 30;
        camera.allowHDR = false; camera.allowMSAA = true;
        float distance = UnityEngine.Mathf.Max(bounds.size.y * 2.05f, bounds.size.x * 2.6f);
        camera.transform.position = bounds.center + new UnityEngine.Vector3(0, .08f, -distance);
        camera.transform.LookAt(bounds.center);
        preview.lights[0].intensity = 1.6f; preview.lights[0].color = new UnityEngine.Color(1, .88f, .73f);
        preview.lights[0].transform.rotation = UnityEngine.Quaternion.Euler(35, 35, 0);
        preview.lights[1].intensity = 1.25f; preview.lights[1].color = new UnityEngine.Color(.65f, .81f, 1);
        preview.lights[1].transform.rotation = UnityEngine.Quaternion.Euler(320, 210, 0);
        preview.ambientColor = new UnityEngine.Color(.42f, .44f, .49f);
        preview.BeginPreview(new UnityEngine.Rect(0, 0, 720, 960), UnityEngine.GUIStyle.none);
        preview.Render(true, true);
        var rendered = preview.EndPreview();
        var rt = UnityEngine.RenderTexture.GetTemporary(720, 960, 0, UnityEngine.RenderTextureFormat.ARGB32);
        var previous = UnityEngine.RenderTexture.active;
        try
        {
            UnityEngine.Graphics.Blit(rendered, rt); UnityEngine.RenderTexture.active = rt;
            pixels = new UnityEngine.Texture2D(720, 960, UnityEngine.TextureFormat.RGBA32, false);
            pixels.ReadPixels(new UnityEngine.Rect(0, 0, 720, 960), 0, 0); pixels.Apply();
            var file = folder + "/loadout-" + state + ".png";
            System.IO.File.WriteAllBytes(file, pixels.EncodeToPNG());
            outputs.Add(new { state = state, file = file, armor = System.Linq.Enumerable.ToArray(System.Linq.Enumerable.Select(equipped, e => e.Id)), weapon = weaponItem.Id, sheathed = true });
        }
        finally { UnityEngine.RenderTexture.active = previous; UnityEngine.RenderTexture.ReleaseTemporary(rt); }
    }
    finally { if (pixels != null) UnityEngine.Object.DestroyImmediate(pixels); if (armor != null) armor.Dispose(); preview.Cleanup(); }
}
return new { files = outputs, scene_before = sceneBefore.path, scene_after = UnityEngine.SceneManagement.SceneManager.GetActiveScene().path, dirty_before = dirtyBefore, dirty_after = UnityEngine.SceneManagement.SceneManager.GetActiveScene().isDirty, play_mode = UnityEditor.EditorApplication.isPlaying };
