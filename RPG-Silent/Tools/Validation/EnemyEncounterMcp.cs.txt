System.Action<EnemyController,float> near=(e,d)=>{
    var filter=new UnityEngine.AI.NavMeshQueryFilter{agentTypeID=1479372276,areaMask=UnityEngine.AI.NavMesh.AllAreas};
    for(int i=0;i<24;i++){
        Vector3 offset=Quaternion.Euler(0f,i*15f,0f)*Vector3.right*d;
        UnityEngine.AI.NavMeshHit navHit;
        if(!UnityEngine.AI.NavMesh.SamplePosition(e.transform.position+offset,out navHit,1.5f,filter))continue;
        Vector3 delta=navHit.position+Vector3.up-e.transform.position-Vector3.up;
        var obstructions=Physics.RaycastAll(e.transform.position+Vector3.up,delta.normalized,delta.magnitude,~0,QueryTriggerInteraction.Ignore);
        if(System.Linq.Enumerable.Any(obstructions,h=>h.collider.GetComponentInParent<RPGSilent.Domain.IDamageable>()==null))continue;
        player().Teleport(navHit.position+Vector3.up*.15f,Quaternion.LookRotation(e.transform.position-navHit.position));return;
    }
    throw new System.Exception("No unobstructed test position around "+e.name);
};

ui.CloseUI("UI/StartUI");

var steps=new System.Collections.Generic.List<System.Func<bool>>();

steps.Add(()=>{
    if(run().State!=DungeonRunState.Running)return false;
    wave=enemies();check(wave.Length==3 && run().TotalEnemies==4,"three initial enemies, four total targets");
    melee=System.Linq.Enumerable.First(wave,e=>e.Definition.Behavior==EnemyBehaviorStyle.Melee);
    caster=System.Linq.Enumerable.First(wave,e=>e.Definition.Behavior==EnemyBehaviorStyle.Ranged);
    foreach(var e in wave)if(e!=melee)e.enabled=false;
    near(melee,7f);started=Time.time;before=damageTaken;return true;
});
steps.Add(()=>{
    sawChase|=melee.State==EnemyState.Chase;
    if(damageTaken==before)return false;
    check(sawChase && melee.HasAggro,"melee detects and chases before attacking");
    check(damageTaken-before==settings.ScaleIncomingDamage(10),"melee natural attack deals one scaled hit");
    started=Time.time;before=damageTaken;return true;
});
steps.Add(()=>{if(Time.time-started<.85f)return false;check(damageTaken==before,"melee recovery does not hit repeatedly");return true;});
steps.Add(()=>{
    if(melee.State!=EnemyState.Attack)return false;
    damage(melee,1);check(melee.State==EnemyState.Hurt && melee.CurrentSkill==null,"ordinary hit interrupts unreleased attack");
    before=damageTaken;started=Time.time;return true;
});
steps.Add(()=>{
    if(Time.time-started<.55f)return false;
    check(damageTaken==before,"interrupted attack has no delayed hit");
    foreach(var e in wave)e.enabled=false;
    caster.enabled=true;near(caster,8f);started=Time.time;before=damageTaken;return true;
});
steps.Add(()=>{
    sawBolt|=UnityEngine.Object.FindObjectsByType<AttackProjectile>(FindObjectsSortMode.None).Length>0;
    if(damageTaken==before)return false;
    check(sawBolt && damageTaken-before==settings.ScaleIncomingDamage(12),"caster fires a traveling projectile that hits player");
    near(caster,1.6f);origin=caster.transform.position;started=Time.time;return true;
});
steps.Add(()=>{
    sawRetreat|=caster.State==EnemyState.Retreat;
    if(!sawRetreat || Vector3.Distance(origin,caster.transform.position)<1f)return false;
    check(sawRetreat,"caster retreats when player closes distance");
    var ex=caster.GetComponent<AttackExecutor>();var ctx=ex.CreateAttack(caster.Definition.Skills[0].Attack,2f,player().transform.position);
    check(!ex.ApplyDamage(melee,ctx,melee.transform.position),"enemy projectile context cannot damage allies");
    check(ctx.Damage==24 && caster.Definition.Skills[0].Attack.Damage==12,"per-instance damage snapshot leaves shared skill unchanged");
    ex.ResetCombat();check(!ex.IsContextValid(ctx),"reset invalidates released attack contexts");
    foreach(var e in wave){e.enabled=true;damage(e,9999);e.Die();}
    check(run().DefeatedEnemies==3 && !run().HasResult && stats.Gold==baselineGold,"wave one counts once and does not settle rewards");
    started=Time.time;return true;
});
steps.Add(()=>{
    boss=run().Boss;if(boss==null)return false;
    check(Time.time-started>=2.3f && boss.MaxHealth==600 && boss.Phase==1,"delayed wave spawns phase-one boss with 600 HP");
    check(run().TotalEnemies==4 && run().DefeatedEnemies==3,"boss wave preserves total progress");
    var view=UnityEngine.Object.FindFirstObjectByType<DungeonUI>();
    check(((GameObject)typeof(DungeonUI).GetField("bossPanel",flags).GetValue(view)).activeInHierarchy,"boss health UI appears");
    near(boss,3f);started=Time.time;before=damageTaken;return true;
});

    if(damageTaken==before)return false;
    check(boss.CurrentSkill!=null && boss.CurrentSkill.RequiredPhase==1,"phase one excludes meteor");
    check(damageTaken-before==settings.ScaleIncomingDamage(18) || damageTaken-before==settings.ScaleIncomingDamage(22),"boss phase-one attack lands");
    damage(boss,310);started=Time.time;return true;
});
steps.Add(()=>{
    sawPhase|=boss.State==EnemyState.PhaseTransition;
    if(boss.Phase!=2 || boss.State==EnemyState.PhaseTransition)return false;
    check(sawPhase && boss.CurrentHealth==290,"half-health threshold triggers phase transition exactly once");
    near(boss,9f);started=Time.time;return true;
});
steps.Add(()=>{
    if(boss.CurrentSkill==null || boss.CurrentSkill.RequiredPhase!=2)return false;
    var warning=System.Linq.Enumerable.FirstOrDefault(UnityEngine.Object.FindObjectsByType<LineRenderer>(FindObjectsSortMode.None),l=>l.name=="Attack warning: "+boss.CurrentSkill.DisplayName);
    if(warning==null)return false;
    sawMeteor=true;before=damageTaken;origin=warning.transform.position;
    log("Meteor telegraph="+origin+" observed player="+player().transform.position);
    damage(boss,1);check(boss.State==EnemyState.Attack,"armored meteor resists ordinary stagger");
    near(boss,Vector3.Distance(origin,boss.transform.position)>6f?3f:9f);started=Time.time;return true;
});
steps.Add(()=>{
    if(Time.time-started<1.8f)return false;
    check(damageTaken==before && Vector3.Distance(origin,player().transform.position)>3f,"locked meteor area can be dodged after telegraph");
    check(boss.Phase==2,"boss remains in phase two");
    log("VISUAL CHECKPOINT: phase-two boss alive; set EC.continue=true to finish");return true;

    if(!(System.AppDomain.CurrentDomain.GetData("EC.continue") is bool) || !(bool)System.AppDomain.CurrentDomain.GetData("EC.continue"))return false;
    damage(boss,9999);boss.Die();started=Time.time;return true;
});
steps.Add(()=>{
    if(!run().HasResult)return false;
    check(run().Succeeded && run().DefeatedEnemies==4 && run().AwardedGold==130 && run().AwardedExp==275 && stats.Gold==baselineGold+130,"all four targets settle 130 gold / 275 XP exactly once");
    check(UnityEngine.Object.FindFirstObjectByType<DungeonUI>().transform.Find("ResultPanel").gameObject.activeInHierarchy,"victory panel visible");

    if(loader.IsLoading || ui.IsUIOpen("UI/LoadingUI") || run()==oldRun || run().State!=DungeonRunState.Running)return false;

    check(stats.CurrentHealth==stats.MaxHealth && player().enabled,"retry revives player");

    check(!DungeonLaunchContext.HasValue && !ui.IsUIOpen(DungeonRunService.UiKey) && !stats.IsDead,"return to town clears encounter UI/context and restores player");

 try{
    if(!UnityEditor.EditorApplication.isPlaying)throw new System.Exception("Play stopped");

 }catch(System.Exception e){cleanup();System.AppDomain.CurrentDomain.SetData("EC.result","FAILED at "+step+": "+e+"\n"+(string)System.AppDomain.CurrentDomain.GetData("EC.log"));}
};

return "Started "+steps.Count+" stages";
