// MCP execute_code body (C# 6), disposable Main Play Mode session.
// Two real dungeon loads; filtered path/terrain checks, then keyboard-driven
// Rigidbody travel and natural enemy chase/attack before a small real AOE kill.
// No player position writes during travel, no asset/PlayerPrefs writes.
if(!UnityEditor.EditorApplication.isPlaying || UnityEngine.SceneManagement.SceneManager.GetActiveScene().name!="Main")throw new System.Exception("Requires Main Play Mode");
if(System.AppDomain.CurrentDomain.GetData("H11.update")!=null)throw new System.Exception("Already running");
var c=GameLifetimeScope.Instance.Container;
var ui=(RPGSilent.Domain.IUIService)c.Resolve(typeof(RPGSilent.Domain.IUIService));
var input=(RPGSilent.Domain.IPlayerInputActions)c.Resolve(typeof(RPGSilent.Domain.IPlayerInputActions));
var loader=(RPGSilent.Domain.ISceneLoader)c.Resolve(typeof(RPGSilent.Domain.ISceneLoader));
var stats=(RPGSilent.Domain.PlayerStats)c.Resolve(typeof(RPGSilent.Domain.PlayerStats));
var session=(PlayerSessionService)c.Resolve(typeof(PlayerSessionService));
var p=session.PlayerRig.GetComponentInChildren<PlayerController>(true);
var db=UnityEngine.Resources.Load<DungeonDatabase>("DungeonDatabase");
var flags=System.Reflection.BindingFlags.Instance|System.Reflection.BindingFlags.NonPublic;
var yaw=typeof(PlayerController).GetField("yaw",flags);
bool background=UnityEngine.Application.runInBackground,lookEnabled=input.Look.enabled;
var savedBindings=UnityEngine.InputSystem.InputActionRebindingExtensions.SaveBindingOverridesAsJson(input.Move);
UnityEngine.InputSystem.InputActionRebindingExtensions.RemoveAllBindingOverrides(input.Move);
UnityEngine.Application.runInBackground=true;
UnityEditor.EditorWindow.GetWindow(typeof(UnityEditor.EditorWindow).Assembly.GetType("UnityEditor.GameView")).Focus();
var keyboard=UnityEngine.InputSystem.InputSystem.AddDevice<UnityEngine.InputSystem.Keyboard>();
System.Action<bool> move=held=>UnityEngine.InputSystem.InputSystem.QueueStateEvent(keyboard,held?new UnityEngine.InputSystem.LowLevel.KeyboardState(UnityEngine.InputSystem.Key.W):new UnityEngine.InputSystem.LowLevel.KeyboardState());
System.Action stop=()=>{move(false);if(p!=null && !p.rb.isKinematic)p.rb.linearVelocity=new UnityEngine.Vector3(0,p.rb.linearVelocity.y,0);};
var profile=UnityEngine.ScriptableObject.CreateInstance<AttackProfile>();profile.DeliveryType=AttackDeliveryType.SelfAoe;profile.Damage=9999;profile.AoeRadius=2;
var disabled=new System.Collections.Generic.List<EnemyController>();
UnityEngine.Events.UnityAction<UnityEngine.SceneManagement.Scene,UnityEngine.SceneManagement.LoadSceneMode> hook=(scene,mode)=>{
 foreach(var root in scene.GetRootGameObjects())foreach(var e in root.GetComponentsInChildren<EnemyController>()){
  // Start configures NavMesh/Rigidbody ownership. Isolating AI before Start must
  // also stop the still-dynamic body from pushing a disabled Agent down slopes.
  e.enabled=false;var body=e.GetComponent<UnityEngine.Rigidbody>();if(body!=null){if(!body.isKinematic)body.linearVelocity=UnityEngine.Vector3.zero;body.isKinematic=true;}disabled.Add(e);
 }
};
UnityEngine.SceneManagement.SceneManager.sceneLoaded+=hook;
foreach(var e in UnityEngine.Object.FindObjectsByType<EnemyController>(UnityEngine.FindObjectsSortMode.None)){e.enabled=false;disabled.Add(e);}
System.Func<DungeonRunService> run=()=> (DungeonRunService)UnityEngine.Object.FindFirstObjectByType<SceneLifetimeScope>().Container.Resolve(typeof(DungeonRunService));
System.Func<string,bool> arrived=name=>UnityEngine.SceneManagement.SceneManager.GetActiveScene().name==name && !loader.IsLoading && !ui.IsUIOpen("UI/LoadingUI") && (name!="Dungeon01" || run().State==DungeonRunState.Running);
var log=new System.Text.StringBuilder();int passed=0;
System.Action<bool,string> check=(ok,label)=>{log.AppendLine((ok?"PASS ":"FAIL ")+label);if(!ok)throw new System.Exception(label);passed++;};
var steps=new System.Collections.Generic.List<System.Func<bool>>();
UnityEngine.AI.NavMeshQueryFilter filter=new UnityEngine.AI.NavMeshQueryFilter();filter.areaMask=UnityEngine.AI.NavMesh.AllAreas;
EnemyController[] enemies=null;EnemyController current=null;UnityEngine.Terrain terrain=null;
UnityEngine.Vector3 home=UnityEngine.Vector3.zero;UnityEngine.Vector3[] corners=null;int corner=0,health=0,frame=0,frames=0;
float walked=0,minEnemyGap=100,minPlayerGap=100;UnityEngine.Vector3 previous=UnityEngine.Vector3.zero;bool moved=false;
int rigId=session.PlayerRig.GetInstanceID(),gold=stats.Gold,exp=stats.Exp;
System.Func<UnityEngine.Vector3,float> gap=pos=>pos.y-(terrain.SampleHeight(pos)+terrain.transform.position.y);
foreach(int entry in new[]{1001,1002}) {
 int id=entry;
 steps.Add(()=>{DungeonLaunchContext.Set(id,DungeonDifficulty.Normal);ui.OpenUI("UI/LoadingUI","Scenes/Dungeon01","UI/MainUI");frame=UnityEngine.Time.frameCount;return true;});
 steps.Add(()=>{
  if(!arrived("Dungeon01") || UnityEngine.Time.frameCount<frame+5)return false;
  input.Look.Disable();terrain=UnityEngine.Terrain.activeTerrain;
  var surface=System.Linq.Enumerable.Single(UnityEngine.Object.FindObjectsByType<Unity.AI.Navigation.NavMeshSurface>(UnityEngine.FindObjectsSortMode.None),s=>s.gameObject.scene==UnityEngine.SceneManagement.SceneManager.GetActiveScene() && s.navMeshData!=null);filter.agentTypeID=surface.agentTypeID;
  enemies=System.Linq.Enumerable.ToArray(System.Linq.Enumerable.OrderBy(UnityEngine.Object.FindObjectsByType<EnemyController>(UnityEngine.FindObjectsSortMode.None),e=>UnityEngine.Vector3.Distance(e.transform.position,p.transform.position)));
  check(enemies.Length==5 && surface.useGeometry==UnityEngine.AI.NavMeshCollectGeometry.PhysicsColliders && surface.buildHeightMesh,id+" persisted collision/height mesh and five targets");
  check(session.PlayerRig.GetInstanceID()==rigId && p.enabled && !p.rb.isKinematic,id+" unique live Rigidbody player");
  UnityEngine.AI.NavMeshHit start;
  check(UnityEngine.AI.NavMesh.SamplePosition(db.GetById(id).SpawnPosition,out start,1,filter),id+" spawn within one metre of correct agent mesh");
  foreach(var e in enemies){
   var agent=e.GetComponent<UnityEngine.AI.NavMeshAgent>();UnityEngine.AI.NavMeshHit end;
   check(agent.isOnNavMesh && agent.agentTypeID==filter.agentTypeID && gap(e.transform.position)>-.12f,id+" "+e.name+" grounded matching agent");
   check(UnityEngine.AI.NavMesh.SamplePosition(e.transform.position,out end,.2f,filter),id+" "+e.name+" authored point on mesh");
   var path=new UnityEngine.AI.NavMeshPath();UnityEngine.AI.NavMesh.CalculatePath(start.position,end.position,filter,path);
   check(path.status==UnityEngine.AI.NavMeshPathStatus.PathComplete,id+" path to "+e.name);
   float lowest=100;int samples=0;
   for(int k=1;k<path.corners.Length;k++){
    int count=UnityEngine.Mathf.Max(1,UnityEngine.Mathf.CeilToInt(UnityEngine.Vector3.Distance(path.corners[k-1],path.corners[k])/.25f));
    for(int j=0;j<=count;j++){UnityEngine.AI.NavMeshHit h;var v=UnityEngine.Vector3.Lerp(path.corners[k-1],path.corners[k],j/(float)count);if(!UnityEngine.AI.NavMesh.SamplePosition(v,out h,2,filter))throw new System.Exception("Path sample off mesh");lowest=UnityEngine.Mathf.Min(lowest,gap(h.position));samples++;}
   }
   check(lowest>-.12f,id+" "+e.name+" path above terrain ("+samples+" samples, min="+lowest+")");
  }
  return true;
 });
 for(int index=0;index<5;index++) {
  int n=index;
  steps.Add(()=>{
   current=enemies[n];check(current!=null && !current.IsDead,id+" target "+n+" alive before encounter");
   stats.RestoreVitals();health=stats.CurrentHealth;home=current.transform.position;
   UnityEngine.AI.NavMeshHit a,b;
   if(!UnityEngine.AI.NavMesh.SamplePosition(p.transform.position,out a,2,filter)||!UnityEngine.AI.NavMesh.SamplePosition(home,out b,.2f,filter))throw new System.Exception("Encounter endpoint off mesh");
   var path=new UnityEngine.AI.NavMeshPath();UnityEngine.AI.NavMesh.CalculatePath(a.position,b.position,filter,path);
   check(path.status==UnityEngine.AI.NavMeshPathStatus.PathComplete,id+" continuous route to encounter "+n);
   corners=path.corners;corner=1;walked=0;previous=p.transform.position;frames=0;minEnemyGap=100;minPlayerGap=100;moved=false;current.enabled=true;return true;
  });
  steps.Add(()=>{
   if(p.IsDead)throw new System.Exception("Unexpected player death");
   frames++;walked+=UnityEngine.Vector3.Distance(previous,p.transform.position);previous=p.transform.position;
   minEnemyGap=UnityEngine.Mathf.Min(minEnemyGap,gap(current.transform.position));minPlayerGap=UnityEngine.Mathf.Min(minPlayerGap,gap(p.transform.position));
   moved|=UnityEngine.Vector3.Distance(current.transform.position,home)>1;
   if(minEnemyGap<-.2f || minPlayerGap<-.2f)throw new System.Exception("Below terrain at "+p.transform.position+" enemy="+current.transform.position);
   if(UnityEngine.Vector3.Distance(p.transform.position,current.transform.position)<1.85f || current.State==EnemyState.Attack){stop();return true;}
   if(corner>=corners.Length)throw new System.Exception("Path exhausted before contact");
   var delta=corners[corner]-p.transform.position;delta.y=0;
   if(delta.magnitude<.45f && corner<corners.Length-1){corner++;return false;}
   float angle=UnityEngine.Quaternion.LookRotation(delta).eulerAngles.y;yaw.SetValue(p,angle);p.transform.rotation=UnityEngine.Quaternion.Euler(0,angle,0);
   move(true);return false;
  });
  steps.Add(()=>{
   stop();minEnemyGap=UnityEngine.Mathf.Min(minEnemyGap,gap(current.transform.position));moved|=UnityEngine.Vector3.Distance(current.transform.position,home)>1;
   if(stats.CurrentHealth==health)return false;
   check(moved && current.GetComponent<UnityEngine.AI.NavMeshAgent>().isOnNavMesh,id+" "+current.name+" natural chase moved over 1m");
   check(minEnemyGap>-.2f && minPlayerGap>-.2f && walked>1,id+" actual input travel + natural hit, walked="+walked.ToString("F1")+", frames="+frames+", min enemy/player="+minEnemyGap.ToString("F3")+"/"+minPlayerGap.ToString("F3"));
   UnityEngine.Physics.SyncTransforms();p.GetComponent<AttackExecutor>().ExecuteProfile(profile);
   check(current.IsDead && run().DefeatedEnemies==n+1,id+" real local AOE defeats exactly encounter "+n);return true;
  });
 }
 steps.Add(()=>{
  check(run().Succeeded && stats.Gold==gold+50 && stats.Exp==exp+125,id+" natural route clears and settles once");gold=stats.Gold;exp=stats.Exp;
  var view=UnityEngine.Object.FindFirstObjectByType<DungeonUI>();((UnityEngine.UI.Button)typeof(DungeonUI).GetField("returnButton",flags).GetValue(view)).onClick.Invoke();return true;
 });
 steps.Add(()=>{if(!arrived("Main"))return false;check(!DungeonLaunchContext.HasValue && session.PlayerRig.GetInstanceID()==rigId && input.GameplayInputEnabled,id+" return restores Main and same player");return true;});
}
int step=0;double deadline=UnityEditor.EditorApplication.timeSinceStartup+100;
UnityEditor.EditorApplication.CallbackFunction tick=null;
System.Action cleanup=()=>{
 UnityEditor.EditorApplication.update-=tick;UnityEngine.SceneManagement.SceneManager.sceneLoaded-=hook;stop();
 if(keyboard!=null && keyboard.added)UnityEngine.InputSystem.InputSystem.RemoveDevice(keyboard);
 UnityEngine.InputSystem.InputActionRebindingExtensions.LoadBindingOverridesFromJson(input.Move,savedBindings,true);
 if(lookEnabled)input.Look.Enable();
 foreach(var e in disabled)if(e!=null && !e.IsDead)e.enabled=true;
 UnityEngine.Object.Destroy(profile);UnityEngine.Application.runInBackground=background;
 System.AppDomain.CurrentDomain.SetData("H11.update",null);
};
tick=()=>{
 try{
  if(!UnityEditor.EditorApplication.isPlaying)throw new System.Exception("Play stopped");
  if(UnityEditor.EditorApplication.timeSinceStartup>deadline)throw new System.TimeoutException("step="+step+" player="+p.transform.position+" corner="+corner+" enemy="+(current==null?"none":current.transform.position.ToString()));
  if(!steps[step]())return;
  step++;deadline=UnityEditor.EditorApplication.timeSinceStartup+100;
  System.AppDomain.CurrentDomain.SetData("H11.result","Running "+step+"/"+steps.Count+" checks="+passed);
  if(step==steps.Count){cleanup();var result="COMPLETE "+passed+" checks\n"+log;System.AppDomain.CurrentDomain.SetData("H11.result",result);System.IO.File.WriteAllText("Temp/CodexNavigationValidation/H11.result.txt",result);}
 }catch(System.Exception ex){cleanup();var result="FAILED "+ex+"\n"+log;System.AppDomain.CurrentDomain.SetData("H11.result",result);System.IO.File.WriteAllText("Temp/CodexNavigationValidation/H11.result.txt",result);}
};
System.AppDomain.CurrentDomain.SetData("H11.update",tick);System.AppDomain.CurrentDomain.SetData("H11.result","Started");UnityEditor.EditorApplication.update+=tick;
return "Started navigation + input travel validation";
