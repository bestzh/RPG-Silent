// MCP execute_code method body. Requires a living idle player in Main.
// Tests real TakeDamage/DeadState with controlled Animator advancement.
// Runtime stats are restored; does not implement gameplay resurrection or save data.
var p=UnityEngine.Object.FindFirstObjectByType<PlayerController>();
if(p==null || p.IsDead || p.GetComponent<PlayerSkillController>().IsCasting)throw new System.Exception("Requires living idle player.");
var stats=(RPGSilent.Domain.PlayerStats)GameLifetimeScope.Instance.Container.Resolve(typeof(RPGSilent.Domain.PlayerStats));
var flags=System.Reflection.BindingFlags.Instance|System.Reflection.BindingFlags.NonPublic;
var a=p.animator;var oldStance=p.CurrentStance;var oldVelocity=p.rb.linearVelocity;bool oldEnabled=p.enabled;
float oldTime=UnityEngine.Time.timeScale;var fields=new[]{"CurrentHealth","CurrentEnergy","Gold","Exp"};
var values=new object[fields.Length];for(int i=0;i<fields.Length;i++)values[i]=typeof(RPGSilent.Domain.PlayerStats).GetField("<"+fields[i]+">k__BackingField",flags).GetValue(stats);
int passed=0;var log=new System.Text.StringBuilder();
System.Action<bool,string> check=(ok,name)=>{log.AppendLine((ok?"PASS ":"FAIL ")+name);if(!ok)throw new System.Exception(name);passed++;};
try{
 UnityEngine.Time.timeScale=0;
 foreach(var stance in new[]{PlayerStance.Relax,PlayerStance.Unarmed,PlayerStance.Armed}){
  foreach(var source in stance==PlayerStance.Relax?new[]{"Idle","Jump"}:new[]{"Idle","Attack","Roll","Jump"}){
   stats.ResetForNewGame();p.enabled=true;p.AnimationController.StopAnimation();p.StateMachine.ChangeState(new IdleState(p));
   p.StanceController.SetStance(stance);a.Rebind();a.SetBool("IsDead",false);a.Play("Base Layer.MoveTree",0,0);a.Update(0);
   var hip=a.GetBoneTransform(UnityEngine.HumanBodyBones.Hips);var standingHip=hip.position;
   if(source=="Attack")p.StateMachine.ChangeState(new AttackState(p));
   else if(source=="Roll")p.StateMachine.ChangeState(new RollState(p));
   else if(source=="Jump")p.StateMachine.ChangeState(new JumpState(p));
   a.Update(.1f);p.rb.linearVelocity=new UnityEngine.Vector3(2,1,3);
   p.TakeDamage(100000);a.Update(0);
   for(int frame=0;frame<18;frame++)a.Update(1f/60f);
   check(p.IsDead&&!p.enabled&&p.StateMachine.CurrentState is DeadState&&a.GetCurrentAnimatorStateInfo(0).IsName("Base Layer.Death"),stance+" death interrupts "+source);
   check(!p.AnimationController.IsPlayingAnimation()&&UnityEngine.Mathf.Abs(p.rb.linearVelocity.x)<.001f&&UnityEngine.Mathf.Abs(p.rb.linearVelocity.z)<.001f,stance+"/"+source+" clears timed action and horizontal velocity");
   typeof(PlayerAnimationController).GetField("currentTimer",flags).SetValue(p.AnimationController,100f);
   typeof(PlayerAnimationController).GetMethod("Update",flags).Invoke(p.AnimationController,null);
   a.SetTrigger("Attack");a.Update(2f);
   check(a.GetCurrentAnimatorStateInfo(0).IsName("Base Layer.Death")&&a.GetBool("IsDead"),stance+"/"+source+" death remains terminal after stale timer and attack trigger");
   if(source=="Idle")check(UnityEngine.Vector3.Distance(standingHip,hip.position)>.25f,stance+" death changes the actual skeleton pose");
  }
 }
 return "COMPLETE "+passed+" checks\n"+log;
}finally{
 for(int i=0;i<fields.Length;i++)typeof(RPGSilent.Domain.PlayerStats).GetField("<"+fields[i]+">k__BackingField",flags).SetValue(stats,values[i]);stats.Refresh();
 p.AnimationController.StopAnimation();p.StanceController.SetStance(oldStance);a.Rebind();a.SetBool("IsDead",false);a.ResetTrigger("Die");a.ResetTrigger("Attack");a.Play("Base Layer.MoveTree",0,0);a.Update(0);
 p.StateMachine.ChangeState(new IdleState(p));p.rb.linearVelocity=oldVelocity;p.enabled=oldEnabled;
 UnityEngine.Time.timeScale=oldTime;
}
